async def _wire(tmp_path, monkeypatch, *plans):
    """The run path with the billing double. The n-th client the API builds
    answers the architect with the n-th plan; the last one repeats."""

    def _make_client():

        # lean validates on the engineering tier, so that reply carries the

        return client

def _messages(client, function: str) -> list[str]:
    """The user messages a client was sent for one function."""
    sent = []
    for call in client.chat_json.call_args_list:
        bound = dict(zip(("function", "system_prompt", "user_message"),
                         call.args), **call.kwargs)
        if bound["function"] == function:
            sent.append(bound["user_message"])
    return sent


def _records(detail: dict, phase: str) -> list[dict]:
    """A run's zero-cost records of one name."""
    records = [p for p in detail["phases"] if p["phase"] == phase]
    assert all(p["cost"] == 0.0 for p in records), records
    return [p["verdict"] for p in records]


async def _sync_run(api_client, workflow="full-team", headers=None) -> dict:

async def _rows(observer) -> int:
    async with observer() as session:

            select(func.count()).select_from(Workflow))).scalar_one()

def _gate_spec(asks="plan.blockers", kind="gate"):
    """A plan and the node after it, which names what it asks."""
    fields = {"gate": {"condition": "plan.ready == true", "on_fail": "blocked",
                       "on_fail_reason": "Plan not ready"},
              "check": {"check": "blockers_named"},
              "ai": {"tier": "architecture", "prompt": "plan"}}[kind]
    return parse({"name": "w", "nodes": [
        {"id": "plan", "kind": "ai", "tier": "architecture", "prompt": "plan"},
        {"id": "plan_ready", "kind": kind, "depends_on": ["plan"],
         "asks_requester": asks, **fields}]})


async def _run_gate(blockers, asks="plan.blockers"):
